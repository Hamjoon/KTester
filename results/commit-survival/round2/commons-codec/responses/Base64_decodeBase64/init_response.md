package org.apache.commons.codec.binary;

import static org.junit.jupiter.api.Assertions.*;

import java.nio.charset.StandardCharsets;
import java.util.Base64 as JdkBase64; // for generating reference data only

import org.junit.jupiter.api.AfterAll;
import org.junit.jupiter.api.AfterEach;
import org.junit.jupiter.api.BeforeAll;
import org.junit.jupiter.api.BeforeEach;
import org.junit.jupiter.api.Test;

/**
 * Unit tests for {@link Base64#decodeBase64(byte[])}.
 *
 * <p>
 * The tests cover typical usage scenarios:
 * <ul>
 *   <li>decoding a normal Base64 encoded byte array</li>
 *   <li>decoding an empty array</li>
 *   <li>decoding a {@code null} reference (expected to return {@code null})</li>
 *   <li>decoding URL‑safe Base64 data (no padding)</li>
 *   <li>decoding data that contains whitespace – the default (lenient) decoder should ignore it</li>
 * </ul>
 * </p>
 */
class Base64_decodeBase64_Test {

    /** Sample plain‑text used for encoding/decoding tests. */
    private static final String SAMPLE_TEXT = "Apache Commons Codec – Base64 test data";

    /** UTF‑8 bytes of {@link #SAMPLE_TEXT}. */
    private byte[] sampleBytes;

    /** Base64 representation of {@link #sampleBytes} generated by the JDK for reference. */
    private byte[] jdkEncoded;

    /** URL‑safe Base64 representation of {@link #sampleBytes} generated by the JDK for reference. */
    private byte[] jdkUrlSafeEncoded;

    @BeforeAll
    static void setupBeforeAll() {
        // No global resources required for these tests.
    }

    @BeforeEach
    void setupBeforeEach() {
        // Prepare data that will be reused by each test method.
        sampleBytes = SAMPLE_TEXT.getBytes(StandardCharsets.UTF_8);
        // JDK's Base64 encoder is used only to obtain a known‑good reference encoding.
        jdkEncoded = JdkBase64.getEncoder().encode(sampleBytes);
        jdkUrlSafeEncoded = JdkBase64.getUrlEncoder().withoutPadding().encode(sampleBytes);
    }

    @AfterEach
    void teardownAfterEach() {
        // Clear instance fields to avoid accidental cross‑test contamination.
        sampleBytes = null;
        jdkEncoded = null;
        jdkUrlSafeEncoded = null;
    }

    @AfterAll
    static void teardownAfterAll() {
        // No global resources to clean up.
    }

    /**
     * Verifies that a standard Base64 encoded byte array is correctly decoded
     * back to the original bytes.
     */
    @Test
    void testDecodeBase64_StandardData() {
        // Encode using the class under test (the static method we are testing).
        byte[] encoded = Base64.encodeBase64(sampleBytes);
        // Decode using the focal method.
        byte[] decoded = Base64.decodeBase64(encoded);
        assertArrayEquals(sampleBytes, decoded, "Decoded bytes should match the original input");
    }

    /**
     * Verifies that decoding an empty byte array yields an empty array
     * (and not {@code null}).
     */
    @Test
    void testDecodeBase64_EmptyArray() {
        byte[] empty = new byte[0];
        byte[] result = Base64.decodeBase64(empty);
        assertNotNull(result, "Result should not be null for empty input");
        assertEquals(0, result.length, "Result length should be zero for empty input");
    }

    /**
     * The static {@code decodeBase64(byte[])} method delegates to a new instance
     * of {@link Base64}. The default instance is lenient and returns {@code null}
     * when the input reference is {@code null}.
     */
    @Test
    void testDecodeBase64_NullInput() {
        assertNull(Base64.decodeBase64(null), "Decoding a null reference should return null");
    }

    /**
     * URL‑safe Base64 strings (without padding) must also be decoded correctly.
     * The default decoder is tolerant of the URL‑safe alphabet.
     */
    @Test
    void testDecodeBase64_UrlSafeNoPadding() {
        // Use the JDK to create a URL‑safe, no‑padding representation.
        byte[] encodedUrlSafe = jdkUrlSafeEncoded;
        // The Apache codec should decode it correctly.
        byte[] decoded = Base64.decodeBase64(encodedUrlSafe);
        assertArrayEquals(sampleBytes, decoded, "URL‑safe decoded bytes should match the original input");
    }

    /**
     * The default decoder is lenient and should ignore whitespace characters
     * such as spaces, line feeds and tabs that appear in the input.
     */
    @Test
    void testDecodeBase64_WithWhitespace() {
        // Insert whitespace into a valid Base64 string.
        String withWs = new String(jdkEncoded, StandardCharsets.US_ASCII)
                .replaceAll("(.{4})", "$1\n\t "); // add newline, tab and space every 4 chars
        byte[] encodedWithWs = withWs.getBytes(StandardCharsets.US_ASCII);
        byte[] decoded = Base64.decodeBase64(encodedWithWs);
        assertArrayEquals(sampleBytes, decoded,
                "Decoder should ignore whitespace and return the original bytes");
    }

    /**
     * When the input contains characters that are not part of the Base64 alphabet,
     * the default (lenient) decoder simply skips them. This test confirms that
     * behaviour by inserting a few illegal characters.
     */
    @Test
    void testDecodeBase64_WithIllegalCharacters() {
        // Take a valid Base64 string and inject some illegal characters.
        String valid = new String(jdkEncoded, StandardCharsets.US_ASCII);
        String polluted = valid.replace('A', '@')   // '@' is not in the Base64 alphabet
                               .replace('B', '#')
                               .replace('C', '$');
        byte[] pollutedBytes = polluted.getBytes(StandardCharsets.US_ASCII);
        // The decoder should ignore the illegal characters and still produce the original bytes.
        byte[] decoded = Base64.decodeBase64(pollutedBytes);
        assertArrayEquals(sampleBytes, decoded,
                "Lenient decoder should ignore non‑Base64 characters and decode the rest");
    }
}
