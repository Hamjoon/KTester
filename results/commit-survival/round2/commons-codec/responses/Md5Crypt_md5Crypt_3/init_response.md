package org.apache.commons.codec.digest;

import org.junit.jupiter.api.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;

import java.util.Random;
import java.util.regex.Pattern;
import java.util.regex.Matcher;
import java.util.Arrays;

import static org.junit.jupiter.api.Assertions.*;

@ExtendWith(MockitoExtension.class)
class Md5Crypt_md5Crypt_3_Test {

    private static final Pattern MD5_CRYPT_PATTERN = Pattern.compile("^\\$1\\$([A-Za-z0-9./]{8})\\$(.+)$");
    private byte[] passwordBytes;

    @BeforeAll
    static void setupBeforeAll() {
        // No global resources required for these tests
    }

    @BeforeEach
    void setupBeforeEach() {
        // a reproducible password for all tests
        passwordBytes = "secret".getBytes(StandardCharsets.UTF_8);
    }

    @AfterEach
    void teardownAfterEach() {
        // Ensure the password buffer is cleared after each test to avoid side‑effects
        if (passwordBytes != null) {
            Arrays.fill(passwordBytes, (byte) 0);
        }
    }

    @AfterAll
    static void teardownAfterAll() {
        // No global cleanup required
    }

    /**
     * Verify that the overload which receives a {@link Random} instance
     * returns a string that conforms to the expected MD5‑crypt format.
     * The salt generated by {@code Random} should be exactly eight characters
     * long and consist of characters from the base‑64 alphabet used by the
     * implementation.
     */
    @Test
    void testMd5CryptWithRandomProducesValidHash() {
        // Use a fixed seed to make the test deterministic
        Random deterministicRandom = new Random(0L);
        String result = Md5Crypt.md5Crypt(passwordBytes.clone(), deterministicRandom);

        assertNotNull(result, "Resulting hash must not be null");

        Matcher m = MD5_CRYPT_PATTERN.matcher(result);
        assertTrue(m.matches(),
                "Hash should match pattern $1$<8‑char‑salt>$<hash>, but was: " + result);

        // Verify that the captured salt is exactly eight characters long
        String salt = m.group(1);
        assertEquals(8, salt.length(),
                "Generated salt length should be 8 characters");
    }

    /**
     * The implementation should throw a {@link NullPointerException}
     * when a {@code null} password array is supplied.
     */
    @Test
    void testMd5CryptNullPasswordThrows() {
        Random rnd = new Random();
        assertThrows(NullPointerException.class,
                () -> Md5Crypt.md5Crypt((byte[]) null, rnd));
    }

    /**
     * The implementation zeroes out the supplied password buffer before returning.
     * This test ensures that behaviour is preserved for the overload that accepts
     * a {@link Random}.
     */
    @Test
    void testMd5CryptZeroesInputBuffer() {
        byte[] buffer = new byte[64];
        Arrays.fill(buffer, (byte) 'A');
        // Clone the buffer to keep an untouched copy for comparison after the call
        byte[] original = buffer.clone();

        Md5Crypt.md5Crypt(buffer, new Random());

        // After the call the original content must be cleared (all zeros)
        assertArrayEquals(new byte[buffer.length], buffer,
                "Password buffer should be zeroed after md5Crypt execution");

        // Ensure that the method actually performed some work (i.e., it didn't just return early)
        // by checking that the returned hash is not null and matches the expected format.
        String hash = Md5Crypt.md5Crypt(original, new Random());
        assertNotNull(hash);
        assertTrue(hash.matches("^\\$1\\$[A-Za-z0-9./]{0,8}\\$.+$"));
    }

    /**
     * Passing a {@code null} {@link Random} instance should result in a
     * {@link NullPointerException} because the implementation attempts to
     * generate a random salt.
     */
    @Test
    void testMd5CryptNullRandomThrows() {
        assertThrows(NullPointerException.class,
                () -> Md5Crypt.md5Crypt(passwordBytes.clone(), (Random) null));
    }
}
